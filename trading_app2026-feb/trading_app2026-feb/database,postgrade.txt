class _PortfolioPerformanceData {
  final String timeframe;
  final double portfolioValue;
  final List<FlSpot> spots;

  _PortfolioPerformanceData({
    required this.timeframe,
    required this.portfolioValue,
    required this.spots,
  });

  // Calculate 24h change by comparing current value with previous day's value
  double get change24h {
    if (spots.isEmpty) return 0;
    // For timeframes longer than 1D, we'll use the last two points
    // For 1D timeframe, we'll use the first and last points
    if (timeframe == "1D") {
      return spots.last.y - spots.first.y;
    } else {
      // For longer timeframes, we'll use the last two points
      if (spots.length >= 2) {
        return spots.last.y - spots[spots.length - 2].y;
      }
      return 0;
    }
  }

  // Calculate total profit by comparing current value with initial investment
  double get totalProfit {
    if (spots.isEmpty) return 0;
    return spots.last.y - spots.first.y;
  }

  // Calculate return percentage
  double get returnPercentage {
    if (spots.isEmpty || spots.first.y == 0) return 0;
    return (totalProfit / spots.first.y) * 100;
  }
}

  String getFormattedTime(String timeframe, double x) {
    switch (timeframe) {
      case "1D":
        final times = [
          "9:30 AM", "10:15 AM", "11:00 AM", "11:45 AM", "12:30 PM",
          "1:15 PM", "2:00 PM", "2:45 PM", "3:30 PM", "4:00 PM"
        ];
        int index = (x * 2).round();
        return index < times.length ? times[index] : times.last;
      case "5D":
        return "Day ${(x / 0.8).ceil()}";
      case "1M":
        return "Week ${(x + 1).round()}";
      case "3M":
        return "Month ${(x / 1.33).ceil()}";
      case "6M":
        return "Month ${(x / 0.67).ceil()}";
      case "YTD":
        final months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep"];
        int index = (x * 2).round();
        return index < months.length ? months[index] : months.last;
      case "1Y":
        final quarters = ["Q1", "Q2", "Q3", "Q4"];
        return quarters[(x).round() % 4];
      case "3Y":
        return "Year ${2021 + (x).round()}";
      case "5Y":
        return "Year ${2019 + (x).round()}";
      case "MAX":
        return "${2019 + (x).round()}";
      default:
        return x.round().toString();
    }
  }
